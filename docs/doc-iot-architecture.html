618/双11 高并发消息限流降级方案

1. 传统旧方案设计

核心目标：在 IoT 设备消息洪峰到来时，保护后端系统不被冲垮。

2. 核心代码实现

（1）消息优先级分级 + 令牌桶限流器

@Component
public class MessageRateLimiter {
    // 高优先级：门锁开锁指令（QOS 2）
    private final RateLimiter highPriorityLimiter = RateLimiter.create(5000.0);
    // 中优先级：门锁状态上报
    private final RateLimiter mediumPriorityLimiter = RateLimiter.create(2000.0);
    // 低优先级：门锁心跳保活
    private final RateLimiter lowPriorityLimiter = RateLimiter.create(500.0);

    public boolean tryAcquire(String topic, String payload) {
        if (topic.contains("lock/cmd")) {
            return highPriorityLimiter.tryAcquire(100, TimeUnit.MILLISECONDS);
        }
        if (topic.contains("lock/state")) {
            return mediumPriorityLimiter.tryAcquire(50, TimeUnit.MILLISECONDS);
        }
        return lowPriorityLimiter.tryAcquire(20, TimeUnit.MILLISECONDS);
    }
}

（2）Sentinel 集成改造 MqttConsumerCallBack

@Component
public class MqttConsumerCallBack implements MqttCallback {
    @Autowired
    private MessageRateLimiter rateLimiter;
    @Autowired
    private MessageHandler handler;

    @Override
    public void messageArrived(String topic, MqttMessage message) {
        // 1. 限流判断
        if (!rateLimiter.tryAcquire(topic, new String(message.getPayload()))) {
            log.warn("限流降级：topic={} 触发限流，消息暂存待处理", topic);
            // 写入本地缓存或 Kafka 死信队列，不丢弃
            deadLetterQueue.send(topic, message);
            return;
        }

        // 2. Sentinel 资源保护
        Entry entry = null;
        try {
            entry = SphU.entry("mqtt-message-arrived");
            handler.handle(topic, new String(message.getPayload()));
        } catch (BlockException e) {
            log.warn("Sentinel 限流/熔断触发：{}", e.getMessage());
            // 降级处理：写入 Kafka 待后续消费
            kafkaTemplate.send("iot-dead-letter", topic, message.getPayload());
        } finally {
            if (entry != null) entry.exit();
        }
    }
}

（3）线程池弹性扩容

@Configuration
public class ThreadPoolConfig {
    @Bean
    public ThreadPoolExecutor iotMessageExecutor() {
        return new ThreadPoolExecutor(
            20, 200, // 核心线程数 20，最大 200（可动态扩容）
            60, TimeUnit.SECONDS,
            new LinkedBlockingQueue<>(2000),
            new ThreadPoolExecutor.CallerRunsPolicy()
        );
    }
}

3. 质变级方案：削峰填谷 + 背压

📌 核心认知
上一版方案本质是"应用层补丁"——在 Consumer 里加 RateLimiter 硬限流，这在真正的大规模 IoT 场景下是下策。业界经过多年验证的质变级方案是：EMQX + Kafka + TDengine 黄金组合。

📌 一句话总结
不要在你自己的代码里限流。把限流/分流/缓冲的能力交给 EMQX + Kafka 这两个专门为此设计的中间件。你的代码只需要做一件事——按自己能承受的速度消费，消费慢的时候 Kafka 帮你存着，一分都不会丢。

消费者微服务核心代码

@Component
public class LockEventConsumer {
    @KafkaListener(topics = "iot-lock-events", concurrency = "3")
    public void consume(List<ConsumerRecord<String, String>> records) {
        // 批量拉取，每次最多 500 条
        List<LockEvent> events = records.stream()
            .map(r -> JSON.parseObject(r.value(), LockEvent.class))
            .collect(Collectors.toList());

        // 批量写入 TDengine
        tdEngineTemplate.batchInsert(events);

        // 手动提交 offset（消费成功才提交）
        ack.acknowledge();
    }
}

eKuiper 流处理规则（Edge 核心）

-- eKuiper SQL 规则：本地过滤 + 聚合，减少跨区域数据传输
SELECT
    mac,
    avg(rssi) as avg_rssi,
    count(*) as event_count,
    window_end() as window_end
FROM "lock/+/event"
WHERE rssi < -70
GROUP BY mac, TUMBLINGWINDOW(ss, 60)
HAVING count(*) > 10
