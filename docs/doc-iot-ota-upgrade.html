升级包管理

1. 升级包数据模型

CREATE TABLE ota_firmware (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    firmware_code   VARCHAR(64)  NOT NULL UNIQUE,  -- 固件编码
    product_model   VARCHAR(32)  NOT NULL,           -- 产品型号
    version         VARCHAR(32)  NOT NULL,           -- 版本号（语义化）
    file_url        VARCHAR(512) NOT NULL,           -- OSS 下载地址
    file_size       INT          NOT NULL,           -- 文件大小（字节）
    file_md5        VARCHAR(32)  NOT NULL,           -- MD5 校验值
    file_sha256     VARCHAR(64)  NOT NULL,           -- SHA256 校验值
    upgrade_type    TINYINT      DEFAULT 1,          -- 1:整包 2:差分包
    min_version     VARCHAR(32),                       -- 最低可升级版本
    description     TEXT,                               -- 更新说明
    status          TINYINT      DEFAULT 0,          -- 0:待发布 1:已发布 2:已停用
    tenant_id       BIGINT       NOT NULL,           -- 租户ID
    created_time    DATETIME     DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_product (product_model, version)
);

2. 升级包上传与校验流程

1📤 上传管理端上传固件包至 OSS，获取 file_urlOSS
2🔐 校验后端计算 MD5 + SHA256，与上传者提供的校验值比对完整性
3📝 入库写入 ota_firmware 表，状态=待发布待审核
4✅ 发布审核通过后状态=已发布，可被升级任务引用已发布

升级任务与灰度推送

1. 升级任务模型

CREATE TABLE ota_upgrade_task (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    task_name       VARCHAR(128) NOT NULL,
    firmware_id     BIGINT       NOT NULL,           -- 关联固件包
    upgrade_strategy TINYINT     NOT NULL,           -- 1:全量 2:灰度 3:定向
    gray_ratio      DECIMAL(5,2) DEFAULT 100,         -- 灰度比例（%）
    gray_stage      TINYINT      DEFAULT 1,          -- 当前灰度阶段
    target_scope    JSON,                               -- 升级范围：租户/门店/设备
    scheduled_time  DATETIME,                          -- 定时升级时间
    status          TINYINT      DEFAULT 0,          -- 0:待执行 1:灰度中 2:全量中 3:已完成 4:已暂停
    tenant_id       BIGINT       NOT NULL,
    created_time    DATETIME     DEFAULT CURRENT_TIMESTAMP
);

2. 灰度推送策略

📌 灰度推送核心原则
灰度推送不是简单的"按比例随机选设备"，而是分阶段、可观察、可回滚的渐进式发布。每个阶段需设置成功率阈值，低于阈值自动暂停。
灰度阶段推送比例观察时长成功率阈值说明
Stage 11%2 小时≥ 98%内部测试设备
Stage 25%4 小时≥ 97%小范围验证
Stage 320%8 小时≥ 95%中等规模验证
Stage 4100%——全量推送

灰度推送核心代码

@Service
public class OtaGrayPushService {

    public void executeGrayStage(OtaUpgradeTask task) {
        // 1. 获取目标设备列表
        List<String> targetDevices = resolveTargetDevices(task);

        // 2. 按灰度比例选取设备
        int grayCount = (int) (targetDevices.size() * task.getGrayRatio() / 100.0);
        List<String> grayDevices = targetDevices.stream()
            .sorted(Comparator.randomOrder())  // 随机打散
            .limit(grayCount)
            .collect(Collectors.toList());

        // 3. 通过 MQTT 下发升级通知
        for (String deviceId : grayDevices) {
            OtaNotifyPayload payload = OtaNotifyPayload.builder()
                .firmwareCode(task.getFirmware().getFirmwareCode())
                .version(task.getFirmware().getVersion())
                .fileUrl(task.getFirmware().getFileUrl())
                .fileSize(task.getFirmware().getFileSize())
                .fileMd5(task.getFirmware().getFileMd5())
                .fileSha256(task.getFirmware().getFileSha256())
                .build();

            mqttClient.publish(
                "/hotel/lock/" + deviceId + "/ota/notify",
                JSON.toJSONString(payload).getBytes(),
                2,   // QoS 2：升级通知恰好一次
                false
            );
        }

        // 4. 记录推送记录
        batchInsertUpgradeRecord(task.getId(), grayDevices);
    }
}

设备端升级执行

1. 设备端升级流程

1📥 接收通知MQTT 收到 OTA 通知，解析升级包信息MQTT QoS 2
2🔍 版本校验检查当前版本是否在可升级范围内前置校验
3⬇️ 下载固件HTTPS 下载固件包，支持断点续传HTTPS
4🔐 完整性校验MD5 + SHA256 双重校验，确保包完整完整性
5💾 写入备份区写入 Flash 备份分区，不覆盖当前固件A/B 分区
6🔄 重启切换设置启动标志位，重启从新分区启动原子切换
7✅ 自检上报启动后自检，成功则上报，失败则回滚自检

2. A/B 分区与原子切换

📌 A/B 分区原理
设备 Flash 分为 Slot A（当前运行）和 Slot B（备份升级）两个分区。升级时固件写入非活动分区，重启时通过 Bootloader 原子切换活动分区。如果新固件启动失败，Bootloader 自动回退到旧分区。

设备端升级状态机

// 设备端升级状态机
enum class OtaState {
    IDLE,               // 空闲
    DOWNLOADING,        // 下载中
    DOWNLOAD_PAUSED,    // 下载暂停（断点续传）
    VERIFYING,          // 校验中
    WRITING,            // 写入 Flash
    REBOOTING,          // 重启切换
    SELF_TEST,          // 自检中
    SUCCESS,            // 升级成功
    ROLLING_BACK,       // 回滚中
    FAILED              // 升级失败
}

// 每个状态变更都通过 MQTT 上报
fun reportState(state: OtaState, progress: Int, message: String?) {
    val payload = jsonEncode(mapOf(
        "mac" to deviceId,
        "taskId" to currentTaskId,
        "state" to state.name,
        "progress" to progress,
        "message" to (message ?: ""),
        "timestamp" to System.currentTimeMillis()
    ))
    mqttClient.publish("/hotel/lock/ota/report", payload, QOS_1, false)
}

升级状态上报与监控

1. 状态上报消费链路

@Component
public class OtaReportConsumer {

    @KafkaListener(topics = "iot-ota-report", concurrency = "3")
    public void consume(List<ConsumerRecord<String, String>> records,
                       Acknowledgment ack) {
        for (ConsumerRecord<String, String> r : records) {
            OtaReport report = JSON.parseObject(r.value(), OtaReport.class);

            // 更新升级记录状态
            otaUpgradeRecordMapper.updateState(
                report.getTaskId(), report.getMac(), report.getState()
            );

            // 关键状态触发灰度推进检查
            if (report.getState() == "SUCCESS"
                || report.getState() == "FAILED") {
                grayPushService.checkAndAdvance(report.getTaskId());
            }
        }
        ack.acknowledge();
    }
}

2. 灰度自动推进

@Service
public class OtaGrayAdvanceService {

    public synchronized void checkAndAdvance(long taskId) {
        OtaUpgradeTask task = taskMapper.selectById(taskId);

        // 统计当前阶段升级结果
        OtaStageStats stats = calculateStageStats(taskId, task.getGrayStage());

        double successRate = (double) stats.success / stats.total;

        if (successRate >= task.getSuccessThreshold()) {
            // 成功率达标 → 推进到下一阶段
            int nextStage = task.getGrayStage() + 1;
            if (nextStage > 4) {
                task.setStatus(TaskStatus.COMPLETED);
            } else {
                task.setGrayStage(nextStage);
                task.setGrayRatio(getStageRatio(nextStage));
                grayPushService.executeGrayStage(task);
            }
        } else if (stats.total >= getMinSampleSize(task.getGrayStage())) {
            // 成功率不达标且样本量足够 → 自动暂停
            task.setStatus(TaskStatus.PAUSED);
            log.warn("灰度暂停：taskId={}, stage={}, successRate={}",
                taskId, task.getGrayStage(), successRate);
            // 发送告警通知
            alertService.sendOtaAlert(task, successRate);
        }
        taskMapper.updateById(task);
    }
}

升级失败回滚

1. 三级回滚机制

🔄 L1 设备自回滚
Bootloader 自动回滚
新固件自检失败
Bootloader 切回旧分区
无需云端介入
☁️ L2 云端指令回滚
MQTT 下发回滚指令
云端检测到批量失败
下发回滚指令
设备切回旧版本
🛑 L3 紧急停推
暂停升级任务
灰度成功率低于阈值
自动暂停 + 告警
人工介入决策

2. 云端指令回滚

@Service
public class OtaRollbackService {

    public void emergencyRollback(long taskId) {
        // 1. 暂停升级任务
        OtaUpgradeTask task = taskMapper.selectById(taskId);
        task.setStatus(TaskStatus.ROLLING_BACK);
        taskMapper.updateById(task);

        // 2. 查找已升级成功的设备，下发回滚指令
        List<String> upgradedDevices = recordMapper.selectSuccessDevices(taskId);

        for (String deviceId : upgradedDevices) {
            OtaRollbackPayload payload = OtaRollbackPayload.builder()
                .targetVersion(task.getFirmware().getMinVersion())  // 回滚到上一个版本
                .reason("emergency_rollback")
                .build();

            mqttClient.publish(
                "/hotel/lock/" + deviceId + "/ota/rollback",
                JSON.toJSONString(payload).getBytes(),
                2, false
            );
        }

        // 3. 发送告警
        alertService.sendRollbackAlert(task, upgradedDevices.size());
    }
}
⚠️ 回滚注意事项
回滚操作本身也有风险——如果旧版本存在安全漏洞，回滚后设备将暴露在风险中。因此回滚前需评估旧版本的安全性，必要时先推送安全补丁再回滚。同时，回滚指令使用 QoS 2 确保恰好一次投递。
📌 一句话总结
OTA 升级的本质是受控变更：灰度推送控制变更范围，A/B 分区保证变更原子性，自动回滚兜底变更失败，状态上报实现变更可观测。四个机制缺一不可。
